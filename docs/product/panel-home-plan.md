# Panel Home Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The Claude panel starts Claude Code or Codex in the founder's data folder, with the app as a tool it can reach but not edit, and the rules written into that folder on every start.

**Architecture:** A pure `panelHome()` decides the working directory (data folder after setup, repo before). `store/agent-home.ts` writes generated `CLAUDE.md` and `AGENTS.md` into the data folder (header plus the repo's `AGENTS.md`) and keeps them gitignored there. `terminal/guardrails.ts` builds a per-launch Claude settings file that denies edits to the app's files and `git push`. The terminal session resolves home per spawn and the server reports it to the panel, which shows "in <folder>" and a pre-setup note. The Import a folder page types its prompt into the panel with full paths.

**Tech Stack:** Next 16, React 19, TypeScript, Vitest (`pnpm test` from `web/`), node-pty behind the terminal server (`web/src/terminal/`), Zod schemas in `web/src/store/schemas.ts`.

Spec: `docs/product/panel-home-direction.md`. Read it first.

## Global Constraints

- Everything under `web/` runs from `web/`: `pnpm test`, `pnpm exec tsc --noEmit`, `pnpm lint`, `pnpm -s check-data`.
- `web/src/store/` is the only code that reads or writes founder-data. The generated files are written by `web/src/store/agent-home.ts`; the terminal and setup call it.
- The data folder may be inside the repo (`<repo>/founder-data`, the default) or anywhere else (`founderive.local.json`, or `FOUNDER_DATA_DIR`). Deny rules must never cover the data folder.
- Claude Code permission rules: file rules use `Edit(...)` (never `Write`); an absolute path starts with `//`; `Bash(git push *)` and `Bash(git push:*)` deny a push. `--settings <file>` overrides keys for the session. `--add-dir <dir>` grants file access and loads that directory's skills.
- Generated file names in the data folder: exactly `CLAUDE.md` and `AGENTS.md`. Their header's first line is exactly `<!-- Generated by Founderive on every start. Don't edit; edit AGENTS.md in the app repo instead. -->`.
- Home modes: `record` (cwd is the data folder; only when `founder.md` exists with `setupComplete: true`) and `repo` (cwd is the repo root).
- Panel copy: running status `Running · claude · in <folder name>`; pre-setup note `Finish setup, then Claude starts in your record.`
- The Windows panel stays unavailable; nothing here changes that.
- Commit after every task, on branch `feat/panel-home` (exists, holds the brief). Commit messages end with the two attribution lines this session uses:

  ```
  Co-Authored-By: Claude Fable 5.1 <noreply@anthropic.com>
  Claude-Session: https://claude.ai/code/session_01KMeQ9eUVyb3vodLLu7B8BM
  ```

```ts
import { readFile, writeFile } from "node:fs/promises";
import path from "node:path";
import { dataDir, repoRoot } from "./paths";
import { writeFileAtomic } from "./write";

/** The two files the panel's agent reads in the data folder. Generated on every start; never records. */
export const AGENT_HOME_FILES = ["CLAUDE.md", "AGENTS.md"] as const;

const HEADER_LINE = "<!-- Generated by Founderive on every start. Don't edit; edit AGENTS.md in the app repo instead. -->";

/**
 * The text of both generated files: a header naming the data folder and the app, then the repo's rules verbatim.
 * Claude Code reads CLAUDE.md, Codex reads AGENTS.md; they're identical so the two agents follow the same rules.
 */
export function agentHomeFiles(rules: string, repo: string, data: string): { claude: string; agents: string } {
  const web = path.join(repo, "web");
  const header = [
    HEADER_LINE,
    "# Founderive panel",
    "",
    `This folder is the founder's data folder (\`${data}\`). The Founderive app is at \`${repo}\`. Run its commands from \`${web}\` (for example \`pnpm -s data-dir\`, \`pnpm -s today --json\`, \`pnpm -s check-data\`). You are in the founder's record; the app is a tool. Don't edit the app's files.`,
    "",
    "",
  ].join("\n");
  const text = `${header}${rules.trimEnd()}\n`;
  return { claude: text, agents: text };
}

/** Adds the names to `<dir>/.gitignore` when they're missing (creating the file if needed). Returns whether it wrote. */
export async function ensureIgnored(dir: string, names: readonly string[]): Promise<boolean> {
  const file = path.join(dir, ".gitignore");
  let current = "";
  try {
    current = await readFile(file, "utf8");
  } catch {
    current = "";
  }
  const listed = new Set(current.split(/\r?\n/).map((l) => l.trim()));
  const missing = names.filter((n) => !listed.has(n));
  if (missing.length === 0) return false;
  const separator = current === "" || current.endsWith("\n") ? "" : "\n";
  await writeFile(file, `${current}${separator}${missing.join("\n")}\n`, "utf8");
  return true;
}

/** Writes CLAUDE.md and AGENTS.md into the data folder from the repo's AGENTS.md, and keeps them out of the data folder's git. */
export async function writeAgentHome(): Promise<void> {
  const repo = repoRoot();
  const data = dataDir();
  const rules = await readFile(path.join(repo, "AGENTS.md"), "utf8");
  const files = agentHomeFiles(rules, repo, data);
  await writeFileAtomic("CLAUDE.md", files.claude);
  await writeFileAtomic("AGENTS.md", files.agents);
  await ensureIgnored(data, AGENT_HOME_FILES);
}
```

```ts
import { mkdtemp, readdir, writeFile } from "node:fs/promises";
import os from "node:os";
import path from "node:path";
import type { PanelHome } from "./protocol";

export type Home = PanelHome & { cwd: string };

/** The program's working directory: the founder's record once setup is complete, the repo root before that. */
export function panelHome(input: { repo: string; data: string | null; setupComplete: boolean }): Home {
  if (input.data && input.setupComplete) return { cwd: input.data, mode: "record", name: path.basename(input.data) };
  return { cwd: input.repo, mode: "repo", name: path.basename(input.repo) };
}

/** `//` is Claude Code's spelling of an absolute path in a permission rule (a single slash anchors to the settings file). */
const absoluteRule = (p: string) => `/${p}`;

/**
 * Claude Code settings for one launch: no edits under any of the repo's entries except the ones to keep (the data
 * folder when it lives inside the repo, and the local settings file), and no git push. An entry with a dot in its name
 * is a file; the rest are folders.
 */
export function claudeSettings(repo: string, entries: string[], keep: string[]): string {
  const kept = new Set(keep);
  const deny = entries
    .filter((name) => !kept.has(name))
    .map((name) => (name.includes(".") ? `Edit(${absoluteRule(path.join(repo, name))})` : `Edit(${absoluteRule(path.join(repo, name))}/**)`));
  deny.push("Bash(git push *)", "Bash(git push:*)");
  return `${JSON.stringify({ permissions: { deny } }, null, 2)}\n`;
}

/** Writes the launch settings to a fresh temp folder and returns the file's path. The data folder is never denied. */
export async function writeLaunchSettings(repo: string, data: string): Promise<string> {
  const entries = (await readdir(repo, { withFileTypes: true })).map((e) => e.name);
  const relative = path.relative(repo, data);
  const dataTop = relative && !relative.startsWith("..") && !path.isAbsolute(relative) ? relative.split(path.sep)[0] : null;
  const keep = ["founderive.local.json", ...(dataTop ? [dataTop] : [])];
  const dir = await mkdtemp(path.join(os.tmpdir(), "founderive-panel-"));
  const file = path.join(dir, "claude-settings.json");
  await writeFile(file, claudeSettings(repo, entries, keep), "utf8");
  return file;
}

/** Claude Code's extra arguments: the repo as an added directory when home is the record, and the settings file always. */
export function claudeArgs(home: Home, repo: string, settingsFile: string): string[] {
  return [...(home.mode === "record" ? ["--add-dir", repo] : []), "--settings", settingsFile];
}

/** Quotes one argument for `sh -c`. */
export function shellQuote(s: string): string {
  return `'${s.replace(/'/g, "'\\''")}'`;
}
```

```ts
 * Runs the founder's own CLI through their login shell (so their PATH and sign-in are theirs). It runs in the founder's
 * data folder once setup is complete, with the app repo reachable as an added directory and a settings file that keeps
 * the app's files and git push out of reach; before setup, in the repo root.

export function loginShellLauncher(repo: string, opts: { shell?: string; now?: () => number } = {}): Launcher {
  const shell = opts.shell || process.env.SHELL || "/bin/zsh";
  const now = opts.now ?? Date.now;
  let cache: { at: number; value: Record<Tool, boolean> } | null = null;
  let inFlight: Promise<Record<Tool, boolean>> | null = null;

  const refresh = async (): Promise<Record<Tool, boolean>> => {
    const entries = await Promise.all(TOOLS.map(async (t) => [t, await commandExists(t, shell)] as const));
    const value = Object.fromEntries(entries) as Record<Tool, boolean>;
    cache = { at: now(), value };
    return value;
  };

  const home = async (): Promise<Home> => {
    let data: string | null = null;
    let setupComplete = false;
    try {
      const dir = dataDir();
      if (existsSync(dir)) {
        data = dir;

      }
    } catch {
      // an invalid founderive.local.json: the app itself reports it; the panel falls back to the repo
    }
    return panelHome({ repo, data, setupComplete });
  };

  return {
    command: async (tool, h) => {
      if (h.mode === "record") await writeAgentHome();
      const extra = tool === "claude" ? claudeArgs(h, repo, await writeLaunchSettings(repo, h.mode === "record" ? h.cwd : path.join(repo, "founder-data"))) : [];
      return { file: shell, args: ["-l", "-c", [tool, ...extra.map(shellQuote)].join(" ")] };
    },
    available: () => {
      if (cache && now() - cache.at < AVAILABILITY_CACHE_MS) return Promise.resolve(cache.value);
      if (!inFlight) inFlight = refresh().finally(() => (inFlight = null));
      return inFlight;
    },
    home,
    platform: process.platform === "win32" ? "windows" : "supported",
  };
```

```ts
/** What the panel shows: which message or buttons, and the header's state in words. */
export type PanelView =
  | { kind: "no-server"; status: "Not started" }
  | { kind: "windows"; status: "Not available" }
  | { kind: "connecting"; status: "Connecting" }
  | { kind: "lost"; status: "Unavailable" }
  | { kind: "not-installed"; status: "Not started" }
  | { kind: "start"; status: "Not started" | "Stopped"; stopped: boolean; tools: Tool[]; note: string | null }
  | { kind: "running"; status: `Running · ${Tool}${string}`; tool: Tool };

const BEFORE_SETUP = "Finish setup, then Claude starts in your record.";

export function panelView({ terminal, platform, conn, state, exited }: PanelInput): PanelView {
  if (platform === "windows" || state?.platform === "windows") return { kind: "windows", status: "Not available" };
  if (!terminal) return { kind: "no-server", status: "Not started" };
  if (conn === "refused" || conn === "closed") return { kind: "lost", status: "Unavailable" };
  if (conn === "connecting" || !state) return { kind: "connecting", status: "Connecting" };
  if (state.running) {
    const where = state.home ? ` · in ${state.home.name}` : "";
    return { kind: "running", status: `Running · ${state.running}${where}`, tool: state.running };
  }
  const tools = TOOLS.filter((t) => state.available[t]);
  if (tools.length === 0) return { kind: "not-installed", status: "Not started" };
  const note = state.home?.mode === "repo" ? BEFORE_SETUP : null;
  return exited ? { kind: "start", status: "Stopped", stopped: true, tools, note } : { kind: "start", status: "Not started", stopped: false, tools, note };
}
```
